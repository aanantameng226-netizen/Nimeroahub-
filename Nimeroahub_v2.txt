local function makeDraggable(handle, target, onTap)
    local dragging, moved, startPos, startInput
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, moved = true, false
            startPos, startInput = target.Position, i.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if onTap and not moved then onTap() end
                end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - startInput
            if d.Magnitude > 6 then moved = true end
            if moved then
                target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
            end
        end
    end)
end

    local cam = workspace.CurrentCamera
    local vs = cam and cam.ViewportSize or Vector2.new(800, 400)
    local mx, mw = main.AbsolutePosition.X, main.AbsoluteSize.X

    if mx + mw + 6 + PANEL_W <= vs.X then
        p.Position = UDim2.new(mp.X.Scale, mp.X.Offset + mw + 6, mp.Y.Scale, mp.Y.Offset)
    elseif mx - 6 - PANEL_W >= 0 then
        p.Position = UDim2.new(mp.X.Scale, mp.X.Offset - PANEL_W - 6, mp.Y.Scale, mp.Y.Offset)
    else
        p.Position = UDim2.new(mp.X.Scale, mp.X.Offset, mp.Y.Scale, mp.Y.Offset + main.AbsoluteSize.Y + 6)
    end
end

-- ====== SP (JALAN KE KOORDINAT) ======
local spConn = nil
local carried = nil -- hewan (Tool) yang sedang dibawa
local picked = nil  -- info hewan dunia yang baru diambil
local extraDropCheck = nil -- diisi di bagian auto ambil drop
local function spSpeed() return math.max(speedLevel, 0.01) * SPEED_PER_LEVEL end

local autoOn = false -- toggle "Auto SP saat ambil hewan"
local holdConn, holdInfo, walkBase = nil, nil, nil
local function cancelHold()
    if holdConn then holdConn:Disconnect() holdConn = nil end
    if holdInfo then
        local h = holdInfo.hum
        if h.Parent and h.WalkSpeed == holdInfo.fast then h.WalkSpeed = holdInfo.base end
        holdInfo = nil
    end
end

local function stopSP(reason)
    cancelHold()
    if spConn then spConn:Disconnect() spConn = nil end
    carried = nil
    picked = nil
    local root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if root then root.AssemblyLinearVelocity = Vector3.zero end
    if reason then Notify(reason) end
end

local function carriedGone()
    if not carried then return false end
    if carried.Parent == nil then return true end
    if carried:IsA("Tool") then
        local bp = lp:FindFirstChild("Backpack")
        local inChar = lp.Character and carried:IsDescendantOf(lp.Character)
        local inBp = bp and carried:IsDescendantOf(bp)
        return not (inChar or inBp)
    end
    return false
end

-- pertahankan kecepatan jalan setelah sampai (tidak kembali pelan)
local function holdSpeed(hum, base)
    cancelHold()
    local untilT = tick() + POST_HOLD
    local fast = math.max(base, POST_WALKSPEED)
    holdInfo = {hum = hum, base = base, fast = fast}
    holdConn = RunService.Heartbeat:Connect(function()
        if tick() > untilT or not hum.Parent or hum.Health <= 0 then
            return cancelHold()
        end
        hum.WalkSpeed = fast
    end)
end

-- tinggi tanah (Y) di titik x,z: tembak sinar ke bawah, lewati part yang tidak padat
local function groundY(x, fromY, z, ignore)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local list = table.clone(ignore)
    for _ = 1, 6 do
        params.FilterDescendantsInstances = list
        local hit = workspace:Raycast(Vector3.new(x, fromY, z), Vector3.new(0, -400, 0), params)
        if not hit then return nil end
        if hit.Instance.CanCollide then return hit.Position.Y end
        list[#list + 1] = hit.Instance
    end
    return nil
end

local function startSP()
    stopSP()
    local char = lp.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not root then Notify("Karakter belum siap") return false end
    local baseY = math.min(root.Position.Y, TARGET.Y)
    if hum then walkBase = hum.WalkSpeed end -- kecepatan asli sebelum SP
    local lastPos = root.Position
    -- abaikan karakter sendiri & pemain lain saat mencari tanah
    local ignore = {char}
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= lp and pl.Character then ignore[#ignore + 1] = pl.Character end
    end
    local stand = ((hum and hum.RigType == Enum.HumanoidRigType.R6) and 3
        or ((hum and hum.HipHeight or 2) + root.Size.Y / 2)) + GROUND_MARGIN
    spConn = RunService.Heartbeat:Connect(function(dt)
        local r = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        if not r or lp.Character ~= char then return stopSP("SP batal: karakter berganti") end
        if not autoOn then return stopSP("SP dihentikan (Auto SP OFF)") end

        -- batal kalau terjatuh / terpental / mati / hewan jatuh
        if hum then
            local st = hum:GetState()
            if hum.Health <= 0 or st == Enum.HumanoidStateType.Dead
                or st == Enum.HumanoidStateType.FallingDown
                or st == Enum.HumanoidStateType.Ragdoll then
                return stopSP("SP batal: terjatuh")
            end
        end
        if r.Position.Y < baseY - 25 then return stopSP("SP batal: terjatuh") end
        if (r.Position - lastPos).Magnitude > 12 then return stopSP("SP batal: posisi berubah") end
        if carriedGone() then return stopSP("SP batal: hewan jatuh, ambil lagi") end
        if extraDropCheck and extraDropCheck() then return stopSP("SP batal: hewan jatuh, ambil lagi") end

        local diff = TARGET - r.Position
        local flatDist = Vector3.new(diff.X, 0, diff.Z).Magnitude

        -- sampai: taruh tepat di atas tanah (tidak tembus ke bawah tanah / objek)
        if flatDist <= STOP_DIST then
            local gy = groundY(TARGET.X, math.max(TARGET.Y, r.Position.Y) + 5, TARGET.Z, ignore)
            local finalY = gy and (gy + stand) or TARGET.Y
            r.CFrame = CFrame.new(Vector3.new(TARGET.X, finalY, TARGET.Z)) * (r.CFrame - r.CFrame.Position)
            stopSP("Sampai di tujuan")
            if hum then holdSpeed(hum, walkBase or hum.WalkSpeed) end
            return
        end

        local step = math.min(spSpeed() * dt, diff.Magnitude)
        local newPos = r.Position + diff.Unit * step
        -- selama perjalanan karakter diangkat sedikit dari tanah (dan tidak masuk ke bawah tanah)
        local gy = groundY(newPos.X, math.max(newPos.Y, r.Position.Y) + 5, newPos.Z, ignore)
        if gy and newPos.Y < gy + stand + LIFT_HEIGHT then
            newPos = Vector3.new(newPos.X, gy + stand + LIFT_HEIGHT, newPos.Z)
        end
        r.AssemblyLinearVelocity = Vector3.zero
        r.CFrame = CFrame.new(newPos, newPos + Vector3.new(diff.X, 0, diff.Z))
        lastPos = newPos
    end)
    return true
end

-- ====== AUTO SP SAAT AMBIL HEWAN (1 toggle, semua pendeteksi aktif) ======
local lastTrigger = 0
local loadedAt = tick()
local spawnedAt = tick()
local lastItemAdded, lastItem = 0, nil
local lastPromptAt = 0 -- terakhir kali kamu menekan tombol ambil
local petPromptCheck = nil -- diisi di bagian auto ambil drop
local function recentPick() return tick() - lastPromptAt < 5 end

local function trigger(reason, inst)
    if not autoOn then return end
    if spConn or tick() - lastTrigger < 1.5 then return end
    lastTrigger = tick()
    Notify("Ambil hewan: " .. reason)
    if startSP() then carried = inst end
end

-- 1) Item baru di Backpack / Character
local seen = setmetatable({}, {__mode = "k"})
local function onNewItem(item)
    if not (item:IsA("Tool") or item:IsA("Model")) then return end
    local isNew = not seen[item]
    seen[item] = true
    if isNew then lastItemAdded, lastItem = tick(), item end
    if isNew and tick() - spawnedAt > 3 and recentPick() then trigger(item.Name, item) end
end
local function watch(container)
    for _, it in ipairs(container:GetChildren()) do seen[it] = true end
    container.ChildAdded:Connect(onNewItem)
    container.ChildRemoved:Connect(function(item)
        task.delay(0.2, function()
            local bp = lp:FindFirstChild("Backpack")
            local stillMine = (bp and item:IsDescendantOf(bp)) or (lp.Character and item:IsDescendantOf(lp.Character))
            if not stillMine then seen[item] = nil end -- lepas dari kita: kalau diambil lagi dianggap baru
        end)
    end)
end

-- ====== AUTO AMBIL HEWAN DROP (minimal B / T per detik) ======
-- Cari ProximityPrompt (tombol ambil) pada hewan yang punya teks "$37.1M/s".
-- Hanya hewan >= minimal (bawaan 1B/s) yang diambil, sisanya dilewati.
-- Timer di atas hewan (misal "21.1s") dipakai supaya tidak mengejar hewan yang keburu hilang.
local MULT = {k=1e3, m=1e6, b=1e9, t=1e12, qa=1e15, qi=1e18, sx=1e21, sp=1e24, oc=1e27, no=1e30, dc=1e33}
local SUFS = {{1e33,"Dc"},{1e30,"No"},{1e27,"Oc"},{1e24,"Sp"},{1e21,"Sx"},{1e18,"Qi"},{1e15,"Qa"},{1e12,"T"},{1e9,"B"},{1e6,"M"},{1e3,"K"}}

local function clean(text)
    local t = text:gsub("<[^>]+>", "")
    t = t:gsub(",", "")
    return t
end

local function parseIncome(text)
    if type(text) ~= "string" then return nil end
    local t = clean(text)
    local num, suf = t:match("%$?%s*([%d%.]+)%s*(%a*)%s*/%s*[sS]")
    local n = tonumber(num)
    if not n then return nil end
    local m = (suf == "") and 1 or MULT[suf:lower()]
    if not m then return nil end
    return n * m
end

local function parseTimer(text)
    if type(text) ~= "string" then return nil end
    return tonumber(clean(text):match("^%s*(%d+%.?%d*)%s*[sS]%s*$"))
end

local function fmtIncome(v)
    for _, e in ipairs(SUFS) do
        if v >= e[1] then return ("%.1f%s"):format(v / e[1], e[2]) end
    end
    return tostring(math.floor(v))
end

local function promptPart(prompt)
    local p = prompt.Parent
    if not p then return nil end
    if p:IsA("BasePart") then return p end
    if p:IsA("Attachment") and p.Parent and p.Parent:IsA("BasePart") then return p.Parent end
    local m = p:IsA("Model") and p or p:FindFirstAncestorOfClass("Model")
    return m and (m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart", true))
end

-- cari label pendapatan & timer milik hewan (di-cache supaya ringan)
local infoCache = setmetatable({}, {__mode = "k"})
local function findLabels(prompt)
    local incL, timL
    local root = prompt.Parent
    for _ = 1, 3 do
        if not root or root == workspace or root:IsA("Folder") then break end
        for _, d in ipairs(root:GetDescendants()) do
            if d:IsA("TextLabel") then
                if not incL and parseIncome(d.Text) then incL = d end
                if not timL and parseTimer(d.Text) then timL = d end
            end
        end
        if incL then break end
        root = root.Parent
    end
    return incL, timL
end

local function getInfo(prompt)
    local e = infoCache[prompt]
    local age = e and (tick() - e.t) or 0
    if not e or (e.inc and not e.inc.Parent) or (not e.inc and age > 1.5) or (e.inc and not e.tim and age > 2) then
        local i, t = findLabels(prompt)
        e = {inc = i, tim = t, t = tick()}
        infoCache[prompt] = e
    end
    if not e.inc then return nil end
    local income = parseIncome(e.inc.Text)
    local timer = (e.tim and e.tim.Parent) and parseTimer(e.tim.Text) or nil
    return income, timer
end

petPromptCheck = function(prompt)
    local model = prompt:FindFirstAncestorOfClass("Model")
    if model and (model == lp.Character or isMine(model)) then return false end
    local inc = getInfo(prompt)
    return inc ~= nil
end

local function petName(c)
    if c.prompt.ObjectText ~= "" then return c.prompt.ObjectText end
    return c.model and c.model.Name or c.prompt.Name
end

local autoDropOn, busy = false, false
local minIncome, radius = 1e9, 300 -- minimal 1B/s (B atau T); radius tetap
local pickSpeed = 150 -- kecepatan ambil: sangat cepat (tetap, tidak ada di menu)
local function pickSpeedVal() return math.max(pickSpeed, 0.01) * SPEED_PER_LEVEL end

local function scan(minV, maxDist, checkTimer)
    local r = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    local out = {}
    if not r then return out end
    for prompt in pairs(prompts) do
        local bl = blacklist[prompt]
        if prompt.Parent and prompt.Enabled and not (bl and bl > tick()) then
            local part = promptPart(prompt)
            if part then
                local d = (part.Position - r.Position).Magnitude
                if d <= maxDist then
                    local model = prompt:FindFirstAncestorOfClass("Model")
                    if not (model and (model == lp.Character or isMine(model))) then
                        local inc, timer = getInfo(prompt)
                        if inc and inc >= minV then
                            -- lewati kalau hewan keburu hilang sebelum sampai
                            local reachable = not (checkTimer and timer and timer < d / pickSpeedVal() + 1)
                            if reachable then
                                out[#out + 1] = {prompt = prompt, part = part, dist = d, income = inc, timer = timer, model = model}
                            end
                        end
                    end
                end
            end
        end
    end
    -- paling mahal dulu (T sebelum B), lalu yang terdekat
    table.sort(out, function(a, b)
        if a.income ~= b.income then return a.income > b.income end
        return a.dist < b.dist
    end)
    return out
end

-- jalan cepat (blocking) ke posisi yang terus diperbarui
local function walkTo(getPos, stop, abortFn, timeout)
    local char = lp.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not root then return false end
    local lastPos, t0 = root.Position, tick()
    while true do
        local dt = RunService.Heartbeat:Wait()
        local r = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        if not autoDropOn or not r or lp.Character ~= char then return false end
        if hum and hum.Health <= 0 then return false end
        if (r.Position - lastPos).Magnitude > 40 then return false end -- terpental / dipindah game
        if tick() - t0 > (timeout or 20) then return false end
        if abortFn and abortFn() then return false end
        local diff = getPos() - r.Position
        local dist = diff.Magnitude
        if dist <= stop then return true end
        local newPos = r.Position + diff.Unit * math.min(pickSpeedVal() * dt, dist)
        r.AssemblyLinearVelocity = Vector3.zero
        r.CFrame = CFrame.new(newPos, newPos + Vector3.new(diff.X, 0, diff.Z))
        lastPos = newPos
    end
end

local function pickup(c)
    local prompt, part, model = c.prompt, c.part, c.model
    Notify(("Kejar %s %s/s"):format(petName(c), fmtIncome(c.income)))
    local arrived = walkTo(function() return part.Position + Vector3.new(0, 3, 0) end, 3,
        function() return not prompt.Parent or not prompt.Enabled end, 20)
    if not arrived then
        blacklist[prompt] = tick() + 2
        return
    end

    -- tekan tombol ambil terus tiap frame (tanpa jeda) sampai hewan terambil
    for _ = 1, 10 do
        if fireproximityprompt then
            pcall(fireproximityprompt, prompt)
        else
            pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(prompt.HoldDuration, 0) + 0.1)
                prompt:InputHoldEnd()
            end)
        end
        RunService.Heartbeat:Wait()
        if not prompt.Parent or not prompt.Enabled then break end
    end

    -- cek berhasil diambil
    local ok, mode, t = false, nil, tick()
    while tick() - t < 2.5 do
        RunService.Heartbeat:Wait()
        if tick() - lastItemAdded < 3 then ok, mode = true, "tool" break end
        if not prompt.Parent or not prompt.Enabled or (model and not model.Parent) then
            ok, mode = true, "world"
            break
        end
    end

    if ok then
        if autoOn and not spConn and startSP() then
            if mode == "tool" then
                carried = lastItem -- jatuh = item hilang dari tas / tangan
            else
                picked = {
                    prompt = prompt, name = petName(c), income = c.income, t = tick(),
                    model = (model and model.Parent) and model or nil,
                    expectDisabled = (prompt.Parent ~= nil and not prompt.Enabled),
                }
            end
        end
    else
        blacklist[prompt] = tick() + 3
        Notify("Gagal ambil, cari yang lain")
    end
end

-- deteksi hewan jatuh untuk hewan "dunia" (bukan Tool)
local lastDropScan = 0
extraDropCheck = function()
    if not picked or tick() - picked.t < 1 then return false end
    if picked.model and not picked.model.Parent then return true end
    if tick() - lastDropScan > 0.25 then
        lastDropScan = tick()
        for _, c in ipairs(scan(0, 60)) do
            local sameName = petName(c) == picked.name
            local sameIncome = math.abs(c.income - picked.income) <= picked.income * 0.02
            if sameName and sameIncome and (c.prompt ~= picked.prompt or picked.expectDisabled) then
                return true
            end
        end
    end
    return false
end
